//  bangbang 转向控制,左转为正
void Turn_Angle_BangBang(float angle){
    static uint8_t first_run = 1;
    if(first_run){
        first_run = 0;
        pose.Flag_Gyro_Start = 1;
    }
    int fast_speed = 20; // 快速阶段速度
    int slow_speed = 10; // 慢速修正阶段速度

    MPU6050_PoseUpdate(&pose, 0.01f);
    if(angle > 0){
        if(angle - pose.yaw > 5.0f){
            Motor_SetPWM_L(0);
            Motor_SetPWM_R(fast_speed);
        }else if((angle > pose.yaw) && (angle - pose.yaw <= 5.0f)){
            Motor_SetPWM_L(0);
            Motor_SetPWM_R(slow_speed);
        }else if((pose.yaw > angle) || ((angle > pose.yaw) && (angle - pose.yaw <= 1.0f))){
            Motor_SetPWM_L(0);
            Motor_SetPWM_R(0);
            pose.Flag_Gyro_Start = 0;
        }
    }
}

void PID_GoStraight_Init(void) {
    // 初始化位置环 (外环)
    /*
        输入：目标距离 [-1000mm ~ 1000mm]
        输出：速度值 [-400rpm ~ 400rpm]
    */
    PID_Init(&pid_pos, PID_TYPE_POSITION);
    pid_pos.Kp = 0.40f;
    pid_pos.Ki = 0.07f;
    pid_pos.Kd = 0.01f;
    pid_pos.output_max = 500.0f;
    pid_pos.output_min = -500.0f;
    pid_pos.enable_output_limit = 1;
    pid_pos.enable_integral_limit = 1;
    pid_pos.integral_max = 500.0f;
    pid_pos.integral_min = -500.0f;
    pid_pos.enable_derivative_filter = 1;
    pid_pos.derivative_filter_alpha = 0.7f;

    // 初始化左轮速度环 (内环)
    /*
        输入：目标速度 [-400rpm ~ 400rpm]
        输出：PWM [-50.0f ~ 50.0f]
    */
    PID_Init(&pid_vel_l, PID_TYPE_POSITION); // 推荐增量式，抗饱和好
    pid_vel_l.Kp = 0.25f;
    pid_vel_l.Ki = 0.01f;
    pid_vel_l.Kd = 0.01f;
    pid_vel_l.output_max = 40.0f;
    pid_vel_l.output_min = -40.0f;
    pid_vel_l.enable_output_limit = 1;
    pid_vel_l.enable_integral_limit = 1;
    pid_vel_l.integral_max = 80.0f;
    pid_vel_l.integral_min = -80.0f;
    pid_vel_l.enable_derivative_filter = 1;
    pid_vel_l.derivative_filter_alpha = 0.7f;
    pid_vel_l.enable_integral_separation = 1;
    pid_vel_l.integral_separation_threshold = 200.0f;

    // 初始化右轮速度环 (内环)
    PID_Init(&pid_vel_r, PID_TYPE_POSITION);
    pid_vel_r.Kp = 0.25f;
    pid_vel_r.Ki = 0.01f;
    pid_vel_r.Kd = 0.01f;
    pid_vel_r.output_max = 40.0f;
    pid_vel_r.output_min = -40.0f;
    pid_vel_r.enable_output_limit = 1;
    pid_vel_r.enable_integral_limit = 1;
    pid_vel_r.integral_max = 80.0f;
    pid_vel_r.integral_min = -80.0f;
    pid_vel_r.enable_derivative_filter = 1;
    pid_vel_r.derivative_filter_alpha = 0.7f;
    pid_vel_r.enable_integral_separation = 1;
    pid_vel_r.integral_separation_threshold = 200.0f;
}

void PID_GoStraight_Control(void) {
     static float target_distance; // 目标距离
     target_distance = target;
     // 注意：Encoder_Update 应该在定时器中断里运行（如40ms）
     // 这里直接读取更新后的数据
     float dis_l = encoder[LEFT_MOTOR_INDEX].distance;  // 左轮距离
     float dis_r = encoder[RIGHT_MOTOR_INDEX].distance; // 右轮距离
     float vel_l = encoder[LEFT_MOTOR_INDEX].rpm;      // 左轮速度
     float vel_r = encoder[RIGHT_MOTOR_INDEX].rpm;      // 右轮速度

     // 外环计算 (位置环)
     // 目标：计算出期望的基础速度 (base_velocity)
     // 原理：基于左右轮平均位置的误差
     float current_avg = (dis_l + dis_r) * 0.5f; // 当前平均位置
     pid_pos.target = target_distance;
     pid_pos.feedback = current_avg;
    
     float base_velocity;
     base_velocity = PID_Calculate(pid_pos.target, pid_pos.feedback, &pid_pos);
    
     // 转向修正计算 (差速环)
     // 目标：计算出左右轮的速度差修正量，防止跑偏
     // 原理：如果右轮比左轮走的远(正差)，说明小车右偏，需要让左轮快一点，右轮慢一点
     float dist_diff = dis_r - dis_l; // 正数表示右轮多，负数表示左轮多
     float turn_correction = dist_diff * BALANCE_Kp; // P控制
        
     // 内环计算 (速度环 x 2)
     // 目标：将目标速度转换为 PWM
     // 左轮目标 = 基础速度 + 修正量 (如果右偏，修正量为正，左轮需加速)
     // 右轮目标 = 基础速度 - 修正量 (如果右偏，修正量为正，右轮需减速)
        
     // --- 左轮控制 ---
     pid_vel_l.target = base_velocity + turn_correction;
     pid_vel_l.feedback = vel_l;
     PID_Calculate(pid_vel_l.target, pid_vel_l.feedback, &pid_vel_l);
     pwm_l = (int16_t)pid_vel_l.output;
     Motor_SetPWM_L(pwm_l);
     // --- 右轮控制 ---
     pid_vel_r.target = base_velocity - turn_correction;
     pid_vel_r.feedback = vel_r;
     PID_Calculate(pid_vel_r.target, pid_vel_r.feedback, &pid_vel_r);
     pwm_r = (int16_t)pid_vel_r.output;
     Motor_SetPWM_R(pwm_r);
    
     // E. 结束判断 (到达目标)     // 如果误差很小且速度很低，则停止
     if(fabs(pid_pos.error) < 1.0f && fabs(base_velocity) < 3.0f) {
         Motor_Stop(&pid_vel_l);
         Motor_Stop(&pid_vel_r);
     }
}
