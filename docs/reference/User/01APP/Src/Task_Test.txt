/**
 * @brief 任务初始化
 * @details 初始化任务，包括定时器、编码器、PWM、MPU6050、编码器和OLED。
 * @param None
 * @retval None
 */
void Task_Init(void){
    HAL_TIM_Base_Start_IT(&htim1);
    DWT_Delay_Init();
    Encoder_Init();
    OLED_Init();
    Motor_Init();
    // PID_StraightControl_Init();
    // PID_Turn_Init();
    MPU6050_Init(&pose);
}

/**
 * @brief 任务：数据采集
 * @details 采集传感器数据。
 * @param None
 * @retval None
 */
void Task_Data_Collection(void){
    MPU6050_GetData(&pose);
    Grayscale_Sensor_Read_All();
}

/**
 * @brief 任务：串口1数据采集与发送,并VOFA+上位机通信
 * @details 从串口1接收数据，解析后根据指令进行运动控制，同时将采集到的传感器数据通过串口1发送。
 * @param None
 * @retval None
 */
void Task_Serial_Vofa(void){
    SendUART1();
    ReceiveUART1();
}
